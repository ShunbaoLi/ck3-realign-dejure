								tier = tier_duchy

								is_landless_type_title = no

									tier = tier_kingdom
									is_noble_family_title = no
									is_landless_type_title = no
									is_titular = no
									NOT = { de_jure_liege = scope:recipient.primary_title }
								}

							scope:actor.primary_title.tier >= tier_hegemony

									any_held_title = {
										tier = tier_empire

										is_landless_type_title = no

											is_noble_family_title = no
											is_landless_type_title = no

					}

							tier = tier_kingdom
